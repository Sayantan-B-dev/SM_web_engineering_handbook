### `typeof` Narrowing
```typescript
function process(value: string | number | boolean) {
  if (typeof value === 'string') {
    return value.toUpperCase(); // string
  }
  if (typeof value === 'number') {
    return value.toFixed(2); // number
  }
  return value; // boolean
}
```

### `instanceof` Narrowing
```typescript
function formatDate(date: Date | string) {
  if (date instanceof Date) {
    return date.toISOString(); // Date
  }
  return date; // string
}

class ApiError extends Error {
  constructor(public statusCode: number, message: string) {
    super(message);
  }
}

try { /* ... */ } catch (err) {
  if (err instanceof ApiError) {
    console.log(err.statusCode); // number — narrowed
  }
}
```

### `in` Operator Narrowing
```typescript
type Fish = { swim: () => void };
type Bird = { fly: () => void };

function move(animal: Fish | Bird) {
  if ('swim' in animal) {
    animal.swim(); // Fish
  } else {
    animal.fly(); // Bird
  }
}
```

### Truthiness Narrowing
```typescript
function greet(name: string | null | undefined) {
  if (name) {
    return `Hello, ${name}`; // string (null/undefined/'' excluded)
  }
  return 'Hello, stranger';
}
```

### Discriminated Union Narrowing
```typescript
type Action =
  | { type: 'add'; payload: number }
  | { type: 'remove'; id: string }
  | { type: 'clear' };

function reducer(state: number[], action: Action): number[] {
  switch (action.type) {
    case 'add':
      return [...state, action.payload]; // payload: number
    case 'remove':
      return state.filter(id => id !== action.id); // id: string
    case 'clear':
      return [];
  }
}
```

### User-Defined Type Guards (`is`)
```typescript
interface User { name: string; email: string }

function isUser(value: unknown): value is User {
  return (
    typeof value === 'object' &&
    value !== null &&
    'name' in value &&
    'email' in value &&
    typeof (value as User).name === 'string' &&
    typeof (value as User).email === 'string'
  );
}

function processData(data: unknown) {
  if (isUser(data)) {
    console.log(data.name); // string — narrowed to User
  }
}
```

### Assertion Functions (`asserts`)
```typescript
function assertIsUser(value: unknown): asserts value is User {
  if (!isUser(value)) {
    throw new Error('Not a user');
  }
}

function handle(data: unknown) {
  assertIsUser(data);
  console.log(data.name); // string — narrowed after assertion
}
```

### `satisfies` Operator (TS 4.9+)
```typescript
// ❌ Type annotation widens
const config: Record<string, string | number> = {
  port: 3000,
  host: 'localhost',
};
config.port.toFixed(); // ❌ Error: string | number

// ❌ Without annotation, no validation
const config2 = {
  port: 3000,
  host: 'localhost',
  invalidKey: true, // no error
};

// ✅ satisfies — validates shape but preserves narrow types
const config3 = {
  port: 3000,
  host: 'localhost',
} satisfies Record<string, string | number>;
config3.port.toFixed(); // ✅ number — type preserved

// ✅ Catches invalid values
const config4 = {
  port: 'not a number',
} satisfies Record<string, string | number>; // ✅ no error (string allowed)
```

```typescript
// Real-world: route config
const routes = {
  home: '/',
  about: '/about',
  user: (id: string) => `/users/${id}`,
} satisfies Record<string, string | ((...args: any[]) => string)>;

routes.home;      // string
routes.user('1'); // string — function type preserved
```

### Exhaustiveness Checking
```typescript
type Shape = 
  | { kind: 'circle'; radius: number }
  | { kind: 'square'; side: number };

function area(shape: Shape): number {
  switch (shape.kind) {
    case 'circle':
      return Math.PI * shape.radius ** 2;
    case 'square':
      return shape.side ** 2;
    default:
      const _exhaustive: never = shape; // compile error if a case is missing
      throw new Error(`Unhandled: ${_exhaustive}`);
  }
}
```

### Real-World Deal-Breakers
- **Type assertion instead of narrowing** → `(value as User).name` doesn't validate at runtime; crashes if wrong.
- **`typeof null === 'object'`** → classic JS quirk; always check `value !== null` first.
- **Narrowing lost in callbacks** → narrowing doesn't persist inside async callbacks:
```typescript
function process(value: string | null) {
  if (value) {
    setTimeout(() => {
      value.toUpperCase(); // ❌ Error: value could be null
    }, 100);
  }
}
```
Fix: assign to a `const` after narrowing.
- **Custom type guards can lie** → `function isUser(v: unknown): v is User { return true; }` compiles but is wrong. Type guards are unchecked assertions.
- **`satisfies` doesn't narrow** → it validates but doesn't change the type. Use it to preserve narrow types while checking.
