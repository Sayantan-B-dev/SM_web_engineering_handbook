## 4.9 Discriminated Unions

### Subtopic Tree
- Discriminant property (literal type)
- Narrowing via discriminant
- Exhaustiveness checking
- Nested discriminated unions
- State machines
- Redux-style actions
- API response modeling

### Syntax & Pattern
```typescript
// The discriminant is a literal-typed property shared by all members
type FetchState<T> =
  | { status: 'idle' }
  | { status: 'loading' }
  | { status: 'success'; data: T }
  | { status: 'error'; error: Error };

function renderState<T>(state: FetchState<T>) {
  switch (state.status) {
    case 'idle':
      return 'Ready to fetch';
    case 'loading':
      return 'Loading...';
    case 'success':
      return state.data; // T — TypeScript knows data exists
    case 'error':
      return state.error.message; // Error — TypeScript knows error exists
  }
}
```

### Redux-Style Actions
```typescript
type TodoAction =
  | { type: 'ADD_TODO'; payload: { text: string } }
  | { type: 'TOGGLE_TODO'; payload: { id: string } }
  | { type: 'DELETE_TODO'; payload: { id: string } }
  | { type: 'CLEAR_ALL' };

function todoReducer(state: Todo[], action: TodoAction): Todo[] {
  switch (action.type) {
    case 'ADD_TODO':
      return [...state, { id: crypto.randomUUID(), text: action.payload.text, done: false }];
    case 'TOGGLE_TODO':
      return state.map(t => t.id === action.payload.id ? { ...t, done: !t.done } : t);
    case 'DELETE_TODO':
      return state.filter(t => t.id !== action.payload.id);
    case 'CLEAR_ALL':
      return [];
  }
}
```

### Nested Discriminated Unions
```typescript
type Notification =
  | { type: 'email'; to: string; subject: string; body: string }
  | { type: 'sms'; phone: string; message: string }
  | { type: 'push'; deviceId: string; title: string; data: Record<string, string> };

type NotificationResult =
  | { status: 'sent'; notification: Notification; sentAt: Date }
  | { status: 'failed'; notification: Notification; error: string; retryable: boolean };

function handleResult(result: NotificationResult) {
  if (result.status === 'sent') {
    // result.notification is Notification
    console.log(`Sent ${result.notification.type} at ${result.sentAt}`);
  } else {
    console.log(`Failed: ${result.error}, retryable: ${result.retryable}`);
  }
}
```

### State Machine
```typescript
type AuthState =
  | { state: 'loggedOut' }
  | { state: 'loggingIn'; email: string }
  | { state: 'loggedIn'; user: User; token: string }
  | { state: 'error'; message: string };

type AuthEvent =
  | { type: 'LOGIN'; email: string; password: string }
  | { type: 'LOGIN_SUCCESS'; user: User; token: string }
  | { type: 'LOGIN_FAILURE'; message: string }
  | { type: 'LOGOUT' };

function authReducer(state: AuthState, event: AuthEvent): AuthState {
  switch (state.state) {
    case 'loggedOut':
      if (event.type === 'LOGIN') {
        return { state: 'loggingIn', email: event.email };
      }
      return state;
    case 'loggingIn':
      if (event.type === 'LOGIN_SUCCESS') {
        return { state: 'loggedIn', user: event.user, token: event.token };
      }
      if (event.type === 'LOGIN_FAILURE') {
        return { state: 'error', message: event.message };
      }
      return state;
    case 'loggedIn':
      if (event.type === 'LOGOUT') return { state: 'loggedOut' };
      return state;
    case 'error':
      if (event.type === 'LOGIN') return { state: 'loggingIn', email: event.email };
      return state;
  }
}
```

### Real-World Deal-Breakers
- **Missing discriminant** → union members without a common literal property can't be narrowed.
- **Discriminant not literal** → `{ type: string }` instead of `{ type: 'add' }` — no narrowing possible.
- **Forgetting exhaustiveness check** → new variant added, switch doesn't handle it, silent runtime error. Use `never` check.
- **Overlapping discriminants** → two variants with the same `type` value → ambiguous narrowing.
- **Optional discriminant** → `{ type?: 'add' }` breaks narrowing.
