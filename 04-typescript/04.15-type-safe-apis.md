### Typed Fetch Wrapper
```typescript
type ApiError = {
  status: number;
  message: string;
  code?: string;
};

type ApiResult<T> =
  | { ok: true; data: T }
  | { ok: false; error: ApiError };

async function apiFetch<T>(
  url: string,
  options?: RequestInit
): Promise<ApiResult<T>> {
  try {
    const res = await fetch(url, {
      headers: { 'Content-Type': 'application/json', ...options?.headers },
      ...options,
    });

    if (!res.ok) {
      return {
        ok: false,
        error: { status: res.status, message: res.statusText },
      };
    }

    const data = (await res.json()) as T;
    return { ok: true, data };
  } catch (err) {
    return {
      ok: false,
      error: { status: 0, message: err instanceof Error ? err.message : 'Network error' },
    };
  }
}

// Usage
const result = await apiFetch<User>('/api/user/1');
if (result.ok) {
  console.log(result.data.name); // User — typed
} else {
  console.log(result.error.status); // number — typed
}
```

### Runtime Validation with Zod
```typescript
import { z } from 'zod';

const UserSchema = z.object({
  id: z.number(),
  name: z.string().min(1),
  email: z.string().email(),
  age: z.number().int().positive().optional(),
  role: z.enum(['admin', 'user', 'guest']).default('user'),
});

// Infer TypeScript type from schema
type User = z.infer<typeof UserSchema>;
// { id: number; name: string; email: string; age?: number; role: 'admin' | 'user' | 'guest' }

// Parse and validate at runtime
const result = UserSchema.safeParse(unknownData);
if (result.success) {
  result.data; // User — fully typed
} else {
  result.error.issues; // validation errors
}

// Type-safe API with validation
async function fetchUser(id: string): Promise<User> {
  const res = await fetch(`/api/users/${id}`);
  const data: unknown = await res.json();
  return UserSchema.parse(data); // throws if invalid
}
```

### Type-Safe Environment Variables
```typescript
// env.ts
import { z } from 'zod';

const EnvSchema = z.object({
  NODE_ENV: z.enum(['development', 'production', 'test']),
  API_URL: z.string().url(),
  PORT: z.coerce.number().default(3000),
  DATABASE_URL: z.string(),
  JWT_SECRET: z.string().min(32),
});

export const env = EnvSchema.parse(process.env);

// Usage — fully typed and validated at startup
console.log(env.PORT); // number
console.log(env.API_URL); // string
// env.MISSING_VAR; // ❌ Error at compile time
```

### Type-Safe Routing
```typescript
// Define route map
type Routes = {
  '/': undefined;
  '/users': undefined;
  '/users/:id': { id: string };
  '/posts/:postId/comments/:commentId': { postId: string; commentId: string };
};

// Type-safe navigate function
function navigate<T extends keyof Routes>(
  route: T,
  ...args: Routes[T] extends undefined ? [] : [params: Routes[T]]
): void {
  // implementation
}

navigate('/');                                    // ✅
navigate('/users');                               // ✅
navigate('/users/:id', { id: '123' });            // ✅
// navigate('/users/:id');                        // ❌ missing params
// navigate('/users/:id', { wrong: '123' });      // ❌ wrong params
```

### tRPC-Style End-to-End Types
```typescript
// server.ts
const appRouter = router({
  getUser: publicProcedure
    .input(z.object({ id: z.string() }))
    .query(async ({ input }) => {
      return db.user.findById(input.id); // typed return
    }),
  createPost: publicProcedure
    .input(z.object({ title: z.string(), body: z.string() }))
    .mutation(async ({ input }) => {
      return db.post.create(input);
    }),
});

export type AppRouter = typeof appRouter;

// client.ts — fully typed, no codegen
const user = await trpc.getUser.query({ id: '123' });
// user is typed based on server return
```

### Real-World Deal-Breakers
- **`as T` on fetch response** → no runtime validation; server can return anything. Use Zod/Valibot.
- **Zod schemas out of sync with DB** → schema says `string`, DB has `number` → runtime error.
- **`z.infer` with `default()`** → fields with defaults become required in the output type but optional in input.
- **Environment variables not validated** → missing `DATABASE_URL` crashes at runtime, not build time.
- **Over-validating in hot paths** → Zod parsing on every render/request adds overhead. Cache parsed results.
- **tRPC/GraphQL codegen staleness** → types are generated; if not regenerated after server changes, client uses stale types.
