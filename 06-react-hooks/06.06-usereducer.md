### Syntax
```jsx
const [state, dispatch] = useReducer(reducer, initialState);
```

```jsx
function reducer(state, action) {
  switch (action.type) {
    case 'increment':
      return { count: state.count + 1 };
    case 'decrement':
      return { count: state.count - 1 };
    case 'reset':
      return { count: action.payload };
    default:
      throw new Error(`Unknown action: ${action.type}`);
  }
}

function Counter() {
  const [state, dispatch] = useReducer(reducer, { count: 0 });

  return (
    <>
      <p>{state.count}</p>
      <button onClick={() => dispatch({ type: 'increment' })}>+1</button>
      <button onClick={() => dispatch({ type: 'decrement' })}>-1</button>
      <button onClick={() => dispatch({ type: 'reset', payload: 0 })}>Reset</button>
    </>
  );
}
```

### Lazy Initialization
```jsx
const [state, dispatch] = useReducer(reducer, props.initialArg, init);

function init(initialCount) {
  return { count: initialCount, history: [] };
}
```

### useReducer vs useState

| Use `useState` when... | Use `useReducer` when... |
|------------------------|--------------------------|
| Simple independent values | Multiple related values |
| One or two state variables | Complex state transitions |
| Straightforward updates | Next state depends on previous |
| Simple toggle / counter | State machine-like logic |

### Complex Example: Form with History
```jsx
const initial = { values: { name: '', email: '' }, errors: {}, touched: {} };

function formReducer(state, action) {
  switch (action.type) {
    case 'change':
      return {
        ...state,
        values: { ...state.values, [action.field]: action.value },
      };
    case 'blur':
      return {
        ...state,
        touched: { ...state.touched, [action.field]: true },
      };
    case 'error':
      return { ...state, errors: { ...state.errors, ...action.errors } };
    case 'reset':
      return initial;
    default:
      return state;
  }
}

function Form() {
  const [state, dispatch] = useReducer(formReducer, initial);
  // ...
}
```

### With Context (global state)
```jsx
const CounterContext = createContext();

function CounterProvider({ children }) {
  const [state, dispatch] = useReducer(reducer, { count: 0 });
  return (
    <CounterContext.Provider value={{ state, dispatch }}>
      {children}
    </CounterContext.Provider>
  );
}

function useCounter() {
  const ctx = useContext(CounterContext);
  if (!ctx) throw new Error('useCounter must be used within CounterProvider');
  return ctx;
}
```

### Real-World Deal-Breakers
- **Mutating state in reducer** → must return new state.
- **Async work inside reducer** → reducers must be pure; do async in handlers or effects, then dispatch.
- **Over-using useReducer for trivial state** → `useState` is simpler.
- **Not throwing on unknown actions** → silent failures; add a default that throws.
- **Dispatch inside render** → infinite loop; only dispatch from handlers/effects.
