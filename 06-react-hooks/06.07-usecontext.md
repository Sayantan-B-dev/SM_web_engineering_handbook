### Syntax
```jsx
const ThemeContext = createContext('light');

function App() {
  return (
    <ThemeContext.Provider value="dark">
      <Toolbar />
    </ThemeContext.Provider>
  );
}

function Toolbar() {
  const theme = useContext(ThemeContext);
  return <div className={theme}>Toolbar</div>;
}
```

### Full Example
```jsx
const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);

  const login = async (credentials) => {
    const user = await api.login(credentials);
    setUser(user);
  };

  const logout = () => setUser(null);

  const value = useMemo(() => ({ user, login, logout }), [user]);

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export function useAuth() {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error('useAuth must be used within AuthProvider');
  return ctx;
}

// Usage
function Profile() {
  const { user, logout } = useAuth();
  if (!user) return <Login />;
  return <button onClick={logout}>Logout {user.name}</button>;
}
```

### Performance: Context Re-renders
**Every consumer of a context re-renders when the context value changes**, even if they only use part of the value.

```jsx
// ❌ Every consumer re-renders when `user` or `theme` changes
const value = { user, theme, login, logout };

// ✅ Split into separate contexts
<AuthContext.Provider value={authValue}>
  <ThemeContext.Provider value={themeValue}>
    {children}
  </ThemeContext.Provider>
</AuthContext.Provider>

// ✅ Or memoize value
const value = useMemo(() => ({ user, login }), [user]);
```

### Context + Reducer Pattern
```jsx
const StateContext = createContext();
const DispatchContext = createContext();

function Provider({ children }) {
  const [state, dispatch] = useReducer(reducer, initial);
  return (
    <StateContext.Provider value={state}>
      <DispatchContext.Provider value={dispatch}>
        {children}
      </DispatchContext.Provider>
    </StateContext.Provider>
  );
}

// Consumers that only dispatch don't re-render on state changes
```

### When NOT to Use Context
- **Prop drilling only 1-2 levels** → just pass props.
- **Frequently changing data** (mouse position, form input) → context re-renders all consumers.
- **State that only some branches need** → use component composition instead.

### Real-World Deal-Breakers
- **Creating Context inside a component** → new context every render; hooks inside break.
- **Missing default value** → `useContext(Ctx)` returns `undefined` → runtime error. Use a null check or a sentinel.
- **Over-relying on context** → "context hell" - nested providers 5+ levels deep.
- **Unmemoized value object** → every render triggers all consumers:
```jsx
// ❌ New object every render
<Ctx.Provider value={{ user, login }}>
```
- **Using context for state that doesn't need to be global** → unnecessary re-renders + complexity.
