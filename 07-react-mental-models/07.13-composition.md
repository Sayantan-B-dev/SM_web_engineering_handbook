### Composition via `children`
```jsx
function Card({ title, children }) {
  return (
    <div className="card">
      <h3>{title}</h3>
      {children}
    </div>
  );
}

<Card title="Hello">
  <p>Any content</p>
</Card>
```

### Slots via Named Props
```jsx
function Layout({ header, sidebar, main }) {
  return (
    <div className="layout">
      <div className="header">{header}</div>
      <div className="sidebar">{sidebar}</div>
      <div className="main">{main}</div>
    </div>
  );
}
```

### Compound Components
```jsx
const TabsContext = createContext();

function Tabs({ defaultTab, children }) {
  const [active, setActive] = useState(defaultTab);
  return (
    <TabsContext.Provider value={{ active, setActive }}>
      {children}
    </TabsContext.Provider>
  );
}

function Tab({ id, children }) {
  const { active, setActive } = useContext(TabsContext);
  return <button aria-selected={active === id} onClick={() => setActive(id)}>{children}</button>;
}

Tabs.Tab = Tab;

// Usage
<Tabs defaultTab="a">
  <Tabs.Tab id="a">A</Tabs.Tab>
  <Tabs.Tab id="b">B</Tabs.Tab>
</Tabs>
```

### Composition vs Configuration
```jsx
// ❌ Configuration-heavy
<Modal
  header={<h1>Hi</h1>}
  body={<p>Content</p>}
  footer={<button>OK</button>}
/>

// ✅ Composition - slots are just children
<Modal>
  <Modal.Header><h1>Hi</h1></Modal.Header>
  <Modal.Body><p>Content</p></Modal.Body>
  <Modal.Footer><button>OK</button></Modal.Footer>
</Modal>
```

### Hooks Compose
```jsx
function useUserWithPosts(userId) {
  const user = useUser(userId);
  const posts = usePosts(user?.id);
  return { user, posts };
}
```

### Real-World Deal-Breakers
- **Inheritance in React** → it's not idiomatic; use composition.
- **Deep HOC chains** → "wrapper hell"; prefer hooks.
- **Over-configured props** → 10+ props for a single component → use children/slots.
- **Render props everywhere** → verbose; hooks replace most cases.
- **Compound component APIs** → great, but need typing discipline (TS) to prevent misuse.
