### `<Link>` Component
```tsx
import Link from 'next/link';

<Link href="/about">About</Link>
<Link href={`/blog/${post.slug}`} prefetch={false}>Read more</Link>
```

### Programmatic Navigation
```tsx
'use client';
import { useRouter } from 'next/navigation';

export default function Button() {
  const router = useRouter();
  return (
    <>
      <button onClick={() => router.push('/dashboard')}>Dashboard</button>
      <button onClick={() => router.back()}>Back</button>
      <button onClick={() => router.refresh()}>Refresh</button>
    </>
  );
}
```

### Server-Side Redirect
```tsx
import { redirect } from 'next/navigation';

export default async function Page() {
  const user = await getUser();
  if (!user) redirect('/login');
  return <h1>Welcome</h1>;
}
```

### Hooks
```tsx
'use client';
import { usePathname, useSearchParams, useParams } from 'next/navigation';

function Nav() {
  const pathname = usePathname();         // '/blog/hello'
  const searchParams = useSearchParams(); // URLSearchParams
  const params = useParams();             // { slug: 'hello' }

  return <p>Path: {pathname}, Q: {searchParams.get('q')}</p>;
}
```

### Loading UI
```tsx
// app/blog/loading.tsx - automatically shown while page.tsx loads
export default function Loading() {
  return <BlogSkeleton />;
}
```
Next.js automatically wraps `page.tsx` in a `<Suspense>` boundary, using `loading.tsx` as the fallback.

### Error UI
```tsx
// app/blog/error.tsx - must be a Client Component
'use client';

export default function Error({ error, reset }: {
  error: Error & { digest?: string };
  reset: () => void;
}) {
  return (
    <div>
      <h2>Something went wrong!</h2>
      <button onClick={() => reset()}>Try again</button>
    </div>
  );
}
```

### Real-World Deal-Breakers
- **`<Link>` without `href`** → runtime error. Always provide a valid path.
- **`useRouter` in Server Component** → Client-only hook. Use `redirect()` instead.
- **`loading.tsx` at root of a fast route** → may cause a flash of loading UI on every navigation. Place it at the segment level where it's needed.
- **`error.tsx` not marked `'use client'`** → build error. Error boundaries must be Client Components.
