### Syntax
```jsx
import { useState } from 'react';

function Counter() {
  const [count, setCount] = useState(0);

  return (
    <div>
      <p>Count: {count}</p>
      <button onClick={() => setCount(count + 1)}>+1</button>
      <button onClick={() => setCount(prev => prev - 1)}>-1</button>
    </div>
  );
}
```

### State is a Snapshot
```jsx
function Counter() {
  const [count, setCount] = useState(0);

  function handleClick() {
    setCount(count + 1);
    setCount(count + 1); // ❌ still +1, not +2!
    // Both calls read the same `count` from this render
  }

  function handleClickFixed() {
    setCount(prev => prev + 1);
    setCount(prev => prev + 1); // ✅ +2
  }

  return <button onClick={handleClickFixed}>{count}</button>;
}
```

**Key mental model:** State is a value **frozen for this render**. `setCount` schedules a new render with the new value - it doesn't change `count` in the current closure.

### State Updates are Asynchronous
```jsx
function Form() {
  const [name, setName] = useState('');

  function handleSubmit() {
    setName('Alice');
    console.log(name); // ❌ still '' - state hasn't updated yet
  }

  // ✅ Use the new value directly or in useEffect
  function handleSubmitFixed() {
    const newName = 'Alice';
    setName(newName);
    console.log(newName); // ✅
  }
}
```

### Batching (React 18)
```jsx
// React 18+: ALL updates batched, even in promises, timeouts, event handlers
function handleClick() {
  setCount(c => c + 1);
  setFlag(f => !f);
  setName('Alice');
  // → 1 re-render (batched)
}

// React 17: only batched inside event handlers
// React 18: batched everywhere (automatic batching)

// To opt out (rare):
import { flushSync } from 'react-dom';
flushSync(() => setCount(c => c + 1)); // forces immediate render
flushSync(() => setFlag(f => !f));     // another render
```

### Lazy Initial State
```jsx
// ❌ Runs expensive function on every render
const [data, setData] = useState(expensiveCompute());

// ✅ Only runs once (on first render)
const [data, setData] = useState(() => expensiveCompute());
```

### State Object vs Multiple State Variables
```jsx
// ❌ Object - must spread to update one field
const [user, setUser] = useState({ name: '', email: '', age: 0 });
setUser({ ...user, name: 'Alice' }); // easy to forget a field

// ✅ Separate states - simpler updates
const [name, setName] = useState('');
const [email, setEmail] = useState('');
const [age, setAge] = useState(0);

// ✅ Object when fields change together (form draft)
const [form, setForm] = useState({ name: '', email: '' });
function updateField(key, value) {
  setForm(prev => ({ ...prev, [key]: value }));
}
```

### Derived State (Anti-Pattern)
```jsx
// ❌ Derived state - causes sync bugs
const [firstName, setFirstName] = useState('');
const [lastName, setLastName] = useState('');
const [fullName, setFullName] = useState(''); // duplicates data

useEffect(() => {
  setFullName(`${firstName} ${lastName}`);
}, [firstName, lastName]);

// ✅ Compute during render
const [firstName, setFirstName] = useState('');
const [lastName, setLastName] = useState('');
const fullName = `${firstName} ${lastName}`; // derived, no state
```

### Real-World Deal-Breakers
- **Reading state immediately after setting** → stale value; use local variables or `useEffect`.
- **`setState` in render** → infinite loop. Only set state in event handlers or effects.
- **Mutating state directly** → `state.push(x)` doesn't trigger re-render; use `setState([...state, x])`.
- **Storing derived data in state** → sync bugs; derive during render.
- **Storing props in state** → state won't update when props change; use props directly.
- **Multiple related states updated separately** → inconsistent renders; combine or use reducer.
