### Pattern
```jsx
function UncontrolledInput() {
  const inputRef = useRef(null);

  function handleSubmit() {
    alert(`Value: ${inputRef.current.value}`);
  }

  return (
    <>
      <input ref={inputRef} defaultValue="initial" />
      <button onClick={handleSubmit}>Submit</button>
    </>
  );
}
```

### `defaultValue` vs `value`
```jsx
// Uncontrolled - DOM owns the value
<input defaultValue="hello" />

// Controlled - React owns the value
<input value={text} onChange={e => setText(e.target.value)} />

// ❌ Mixing - React warns
<input value="hello" /> // no onChange → read-only
```

### File Inputs
```jsx
function FileUpload() {
  const fileRef = useRef(null);

  async function upload() {
    const file = fileRef.current.files[0];
    const formData = new FormData();
    formData.append('file', file);
    await fetch('/api/upload', { method: 'POST', body: formData });
  }

  return (
    <>
      <input type="file" ref={fileRef} accept="image/*" />
      <button onClick={upload}>Upload</button>
    </>
  );
}
```

### Integration with Non-React Libraries
```jsx
// Non-React libraries (e.g., third-party editors, maps, charts)
// manage their own DOM. Use refs to read their state.
function RichEditor() {
  const editorRef = useRef(null);

  useEffect(() => {
    const editor = new ThirdPartyEditor(editorRef.current);
    return () => editor.destroy();
  }, []);

  function getContent() {
    // Read from editor instance
    return editorRef.current.innerHTML;
  }

  return <div ref={editorRef} />;
}
```

### Real-World Deal-Breakers
- **Setting `value` on uncontrolled input** → React switches to controlled mode; you'll get a warning.
- **Using `defaultValue` on a controlled input** → ignored after first render.
- **Reading `ref.current.value` before mount** → `null` reference.
- **Controlled-to-uncontrolled switch** → state becomes `undefined` mid-lifecycle → React warning. Keep consistent.
- **File inputs controlled** → throws an error.
