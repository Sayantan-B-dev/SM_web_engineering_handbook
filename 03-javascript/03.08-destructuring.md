### Array Destructuring
```javascript
const [first, second, ...rest] = [1, 2, 3, 4, 5];
console.log(first);  // 1
console.log(second); // 2
console.log(rest);   // [3, 4, 5]

// Skip elements
const [, , third] = [1, 2, 3];
console.log(third); // 3

// Default values
const [a = 10, b = 20] = [1];
console.log(a); // 1
console.log(b); // 20

// Swap variables
let x = 1, y = 2;
[x, y] = [y, x];
console.log(x, y); // 2, 1
```

### Object Destructuring
```javascript
const user = {
  name: 'Alice',
  age: 30,
  address: {
    city: 'Kolkata',
    zip: '700001',
  },
};

// Basic
const { name, age } = user;

// Rename
const { name: userName, age: userAge } = user;

// Default values
const { role = 'user' } = user; // 'user' (not in user)

// Nested
const { address: { city, zip } } = user;

// Rest
const { name: n, ...others } = user;
console.log(others); // { age: 30, address: {...} }

// Combined: rename + default
const { name: fullName = 'Anonymous', country = 'India' } = user;
```

### Function Parameter Destructuring
```javascript
// Object params (named — order-independent)
function createUser({ name, age = 18, role = 'user' }) {
  return { name, age, role };
}

createUser({ name: 'Alice', age: 30 });

// Array params (positional)
function getFirstTwo([first, second]) {
  return [first, second];
}
getFirstTwo([10, 20, 30]); // [10, 20]

// Mixed
function process({ items: [first, ...rest], limit = 10 }) {
  console.log(first, rest, limit);
}
process({ items: [1, 2, 3], limit: 5 });
```

### Real-World Tricks
- **Destructure in function signature** → self-documenting, avoids `options.foo` repetition.
- **Default `{}` for optional object params**:
```javascript
function greet({ name = 'Guest' } = {}) {
  console.log(`Hello, ${name}`);
}
greet(); // "Hello, Guest" — no TypeError
```
Without `= {}`, calling `greet()` throws because you can't destructure `undefined`.

- **Swap without temp variable**: `[a, b] = [b, a]`.
- **Extract from `Promise.all`**:
```javascript
const [users, posts, comments] = await Promise.all([
  fetchUsers(), fetchPosts(), fetchComments()
]);
```

### Real-World Deal-Breakers
- **Destructuring `null`/`undefined`** → `const { name } = null` throws `TypeError`. Use defaults: `const { name } = user ?? {}`.
- **Nested destructuring without defaults** → `const { address: { city } } = user` throws if `address` is undefined.
- **Array destructuring order matters** → `const [a, b] = [b, a]` doesn't swap; it just assigns.
- **Over-destructuring** → pulling 15 properties from an object makes it unclear what's used. Consider passing the object or using a namespace.
