### Object Creation
```javascript
// Literal (preferred)
const user = {
  name: 'Alice',
  age: 30,
  'user-role': 'admin',    // quoted key
  [dynamicKey]: 'value',    // computed key
  greet() { return `Hi, ${this.name}`; },  // shorthand method
};

// Access
user.name;          // dot notation
user['name'];       // bracket notation
user['user-role'];  // bracket required for hyphens

// Dynamic access
const key = 'age';
user[key]; // 30
```

### Getters & Setters
```javascript
const temperature = {
  _celsius: 0,

  get fahrenheit() {
    return this._celsius * 9/5 + 32;
  },

  set fahrenheit(value) {
    this._celsius = (value - 32) * 5/9;
  },
};

temperature.fahrenheit = 100;
console.log(temperature._celsius);    // 37.777...
console.log(temperature.fahrenheit);  // 100
```

### Object Static Methods
```javascript
const obj = { a: 1, b: 2, c: 3 };

Object.keys(obj);    // ['a', 'b', 'c']
Object.values(obj);  // [1, 2, 3]
Object.entries(obj); // [['a', 1], ['b', 2], ['c', 3]]

// fromEntries — reverse of entries
const entries = [['name', 'Alice'], ['age', 30]];
const user = Object.fromEntries(entries); // { name: 'Alice', age: 30 }

// assign — shallow merge
const merged = Object.assign({}, obj, { d: 4 });

// spread — shallow merge (preferred)
const merged2 = { ...obj, d: 4 };
```

### Immutability Methods
```javascript
const obj = { a: 1, nested: { b: 2 } };

Object.freeze(obj);
obj.a = 10;         // silently fails (or throws in strict mode)
obj.nested.b = 20;  // ✅ MUTATES! freeze is shallow
console.log(obj.a);       // 1
console.log(obj.nested.b); // 20

// True deep freeze
function deepFreeze(obj) {
  Object.freeze(obj);
  Object.values(obj).forEach(deepFreeze);
  return obj;
}
```

### Property Descriptors
```javascript
const obj = {};

Object.defineProperty(obj, 'name', {
  value: 'Alice',
  writable: false,      // can't change
  enumerable: false,    // won't show in for...in / Object.keys
  configurable: false,  // can't delete or redefine
});

obj.name = 'Bob';       // silently fails
console.log(obj.name);  // 'Alice'
console.log(Object.keys(obj)); // []
```

### Optional Chaining & Nullish Coalescing
```javascript
const user = {
  name: 'Alice',
  address: {
    city: 'Kolkata',
  },
};

// Optional chaining
user.address?.city;       // 'Kolkata'
user.contact?.phone;      // undefined (no error)
user.getAge?.();          // undefined (method doesn't exist)

// Nullish coalescing: only null/undefined trigger fallback
const count = 0;
console.log(count ?? 10);  // 0 (0 is not null/undefined)
console.log(count || 10);  // 10 (0 is falsy)

const name = null;
console.log(name ?? 'Guest'); // 'Guest'
```

### Real-World Deal-Breakers
- **`||` vs `??`** → `value || default` treats `0`, `''`, `false` as missing. Use `??` when only `null`/`undefined` should trigger the fallback.
- **Shallow copy trap** → `{ ...obj }` copies nested objects by reference. Mutating nested objects affects both.
- **`Object.freeze` is shallow** → nested objects remain mutable.
- **Prototype pollution via `__proto__`** → `Object.assign({}, JSON.parse('{"__proto__":{"admin":true}}'))` can pollute. Use `Object.create(null)` or sanitize.
- **Property ordering** → integer-like keys come first in ascending order, then string keys in insertion order. Don't rely on object key order for logic.
