### The Prototype Chain
Every object has a hidden `[[Prototype]]` link to another object (or `null`). When you access a property, JS walks up the chain until it finds it or reaches `null`.

```javascript
const animal = {
  speak() { console.log(`${this.name} makes a sound`); },
};

const dog = Object.create(animal);
dog.name = 'Rex';
dog.speak(); // "Rex makes a sound" — speak found on prototype

// Prototype chain: dog → animal → Object.prototype → null
console.log(Object.getPrototypeOf(dog) === animal); // true
console.log(Object.getPrototypeOf(animal) === Object.prototype); // true
console.log(Object.getPrototypeOf(Object.prototype)); // null
```

### Constructor Functions & `prototype`
```javascript
function Person(name, age) {
  this.name = name;
  this.age = age;
}

// Methods on prototype — shared across all instances (memory efficient)
Person.prototype.greet = function() {
  console.log(`Hi, I'm ${this.name}`);
};

const alice = new Person('Alice', 30);
const bob = new Person('Bob', 25);

alice.greet(); // "Hi, I'm Alice"
bob.greet();   // "Hi, I'm Bob"

// Both share the same function reference
console.log(alice.greet === bob.greet); // true
```

### `class` Syntax (Syntactic Sugar)
```javascript
class Person {
  constructor(name, age) {
    this.name = name;
    this.age = age;
  }

  greet() {              // on Person.prototype
    console.log(`Hi, I'm ${this.name}`);
  }

  static create(name) {  // on Person (not prototype)
    return new Person(name, 0);
  }
}

class Employee extends Person {
  constructor(name, age, role) {
    super(name, age);    // must call super before `this`
    this.role = role;
  }

  greet() {
    super.greet();       // call parent method
    console.log(`I'm a ${this.role}`);
  }
}
```

### `instanceof` vs `typeof` vs `constructor`
```javascript
const arr = [];
console.log(arr instanceof Array);   // true
console.log(typeof arr);             // 'object'
console.log(arr.constructor === Array); // true

// instanceof checks prototype chain
class Animal {}
class Dog extends Animal {}
const d = new Dog();
console.log(d instanceof Dog);    // true
console.log(d instanceof Animal); // true
console.log(d instanceof Object); // true
```

### `hasOwnProperty` vs `in`
```javascript
const obj = { a: 1 };

console.log('a' in obj);                    // true (own)
console.log('toString' in obj);             // true (inherited)
console.log(obj.hasOwnProperty('a'));       // true
console.log(obj.hasOwnProperty('toString')); // false

// Modern: Object.hasOwn()
console.log(Object.hasOwn(obj, 'a'));       // true
console.log(Object.hasOwn(obj, 'toString')); // false
```

### Real-World Deal-Breakers
- **Mutating `Object.prototype`** → breaks everything globally. Never do `Object.prototype.foo = ...`.
- **`for...in` without `hasOwnProperty`** → iterates inherited enumerable properties.
- **Extending built-ins** → `Array.prototype.myMethod = ...` can conflict with future native methods or libraries.
- **Prototype pollution** → attacker injects `__proto__` via user input, modifying all objects. Sanitize JSON input and avoid recursive merge without guards.
- **`instanceof` across realms** → an array from an iframe is not `instanceof Array` in the parent frame. Use `Array.isArray()` instead.
